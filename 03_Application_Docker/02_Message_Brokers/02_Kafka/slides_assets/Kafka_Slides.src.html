from kafka import KafkaProducer

producer = KafkaProducer(bootstrap_servers='localhost:9092')

# Kafka รับ-ส่งเป็น bytes เสมอ → .encode()  ·  ไม่ต้องสร้าง topic ก่อน (auto-create)
future = producer.send('hello', 'Hello Kafka!'.encode())

# .get() รอ "ใบเสร็จ" RecordMetadata จาก broker
m = future.get(timeout=10)
print(m.topic, m.partition, m.offset)      # → hello 0 0
producer.close()
  • offset 0 = ข้อความแรกสุดของเล่มส่งซ้ำอีก 3 ครั้ง → 1, 2, 3 · ไม่มีเว้น ไม่มีถอยหลัง ไม่รีเซ็ตแม้ broker จะ restart (log เขียนลง disk)
  • ส่งได้แม้ยังไม่มีผู้อ่านสักคนข้อความ "นอน" อยู่ใน log — ผู้ส่งผู้รับไม่ต้องออนไลน์พร้อมกัน
  • send() เป็น async คืน future มาก่อน.get() คือการยืนรอใบเสร็จ · close() ดันของค้างใน buffer ออกให้หมด
auto-create มีราคา : topic ที่ broker สร้างให้เองได้ partition ตามค่า default ของ broker (num.partitions = 1) — และถ้าสะกดชื่อ topic ผิด broker ก็สร้าง "topic ผี" ให้เงียบ ๆ ไม่มี error สักบรรทัด (section 03 จะอธิบายว่าทำไมเรื่องนี้สำคัญ)

# hash_key.py — สูตรเดียวกับที่ producer ใช้
from kafka.partitioner.default import murmur2
for key in ['bangkok','chiangmai','hatyai','korat']:
    h = murmur2(key.encode()) & 0x7fffffff
    print(key, h % 3, h % 4)

chiangmai  hash=  34321306   %3 -> p1   %4 -> p2
hatyai     hash= 431853363   %3 -> p0   %4 -> p3
korat      hash=1798336970   %3 -> p2   %4 -> p2

RabbitMQ vs Kafka — เทียบชัด ๆ

RabbitMQ (ชุดที่แล้ว)Kafka (วันนี้)
เก็บข้อความยังไงqueue — อ่าน + ack แล้วลบทิ้งlog — อ่านแล้วยังอยู่ จนครบ retention
อ่านย้อนหลัง / replayทำไม่ได้ — ข้อความหายไปแล้วได้ตราบที่ยังอยู่ใน retention — group ใหม่เริ่มอ่านจากต้น log

เส้นทางข้อความbroker ฉลาด : exchange + binding + routing keybroker เรียบง่าย : topic + key + group เท่านั้น
ผู้รับได้ข้อความยังไงbroker push (callback)consumer pull (poll)
หน้าเว็บดูสถานะManagement UI ฝังกับ broker (15672)container แยก เช่น Kafka UI (แล็บ : 8411–8414)
เหมาะที่สุดกับคิวงาน — สั่งงานให้ทำครั้งเดียวแล้วจบสายธาร event — หลายทีมใช้ข้อมูลเดียวกัน + ย้อนอดีต
