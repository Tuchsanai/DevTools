ส่วนที่ B — ฟังก์ชันตัวช่วย (วางไว้ต้น session)
--------------------------------------------------------------------------------

```bash
SPEC=/root/workspace/DevTools/05_kubernetes/PROMPT_Kubernetes_Course_Spec.txt
ROOT=/root/workspace/DevTools/05_kubernetes
APP=/root/workspace/DevTools/02_Docker/03_Fullstack_App_Example
SLIDE_REF=$APP/Fullstack_Slides.html

# --- เรียก Claude (yolo) แล้วดึงเฉพาะคำตอบ ---
ask() {  # ask <N> <model> <effort> <prompt>
  "yolo$1" -p --model "$2" --effort "$3" --strict-mcp-config \
    --output-format json --no-session-persistence "$4" < /dev/null \
  | python3 -c "import json,sys;print(str(json.load(sys.stdin).get('result','')).strip())"
}

# --- เรียก Codex (cyolo) แล้วเก็บคำตอบลงไฟล์ ---
do_heavy() {  # do_heavy <outfile> <prompt>
  cyolo1 exec -m gpt-5.6-sol -c model_reasoning_effort="high" \
    --add-dir "$ROOT" --add-dir "$APP" -o "$1" "$2" < /dev/null
}
```

  หมายเหตุ : ตั้ง Bash timeout ของงาน cyolo ไว้ >= 900000 ms เพราะงานถึกใช้เวลานาน
  และเช็คโควตาก่อนเริ่มทุกครั้ง (`usage_all 1 2` และ `cusage_all 1`)
